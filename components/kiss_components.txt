# The parts 'shh config' can turn off (KISS names). Everything else is
# always installed.
#
# name|packages|files (repo paths)|services|group|default
#
# The group is the tab in 'shh config': empty is features, misc is the
# programs we always use, settings is an own line in settings (maxhz).
# xwayland, editor and files are the lines settings > xwayland, editor
# and file manager. recipes is the KISS recipes tab: the packages of our
# own recipes (system/kiss/repo) we can do without. The default is on
# when empty, off when "no".
#
# The packages stay in system/kiss/packages/common.csv too. The sway
# lines of a part go in config/sway/features/NAME.
#
# There is no screenshare: KISS's firefox is built without dbus, so the
# portal can't reach it.

record|wf-recorder|scripts/record config/sway/features/record|||
screenshot|grim|scripts/screenshot config/sway/features/screenshot|||
bluetooth|bluez|scripts/bluetooth config/sway/features/bluetooth|bluetoothd||
clipboard|cliphist|scripts/clipboard config/cliphist config/sway/features/clipboard|||
autotile||scripts/autotile config/sway/features/autotile|||
fastfetch|fastfetch|config/fastfetch|||
scratchpad||config/sway/features/scratchpad|||
# The vpn menu and our own runit service shh-vpn (install.sh runs
# 'vpn setup').
sing-box|sing-box|scripts/vpn config/sway/features/vpn|shh-vpn||

# misc: the programs the setup counts on.
firefox|firefox|config/sway/features/firefox||misc|
mpv|mpv|||misc|
imv|imv|config/imv||misc|
btop|btop|||misc|
7zip|7zip|||misc|
man|mandoc man-pages|||misc|
mako|mako libnotify|config/mako config/sway/features/mako||misc|
emoji|noto-fonts-emoji|||misc|
noto|noto-fonts|||misc|

# settings > xwayland. noxwayland is on when xwayland is off: sway can't
# have both lines.
noxwayland||config/sway/features/noxwayland||xwayland|
# No package: KISS's wlroots and sway are built without Xwayland, so the
# line in settings changes nothing.
xwayland||||xwayland|

# settings > editor: vim or nvim, with the same look, options and keys.
vim|vim|config/vim||editor|
nvim|neovim|config/nvim||editor|

# settings > file manager: broot or yazi on Super+F.
broot|broot|config/broot config/sway/features/broot||files|
yazi|yazi|config/yazi config/sway/features/yazi||files|

# settings > maxhz.
maxhz||scripts/maxhz config/sway/features/maxhz||settings|

# recipes: the packages from system/kiss/repo the setup does not need.
bash|bash|||recipes|
